%%--------------------------------------------------
%% BNF grammar extraction.

% Define the key-value options.
\pgfkeys{
  /bnf/.is family,
  /bnf/term/.initial=missing,
  /bnf/oneof/.initial=false,
  /bnf/noref/.initial=false
}

% Define the base environment for BNF grammar.
\NewDocumentEnvironment{bnfbase}{ O{term=missing,oneof=false,noref=false} m }
{%
  % Set keys using pgfkeys
  \pgfkeys{/bnf/.cd,#1}%
  % Check if "one of" should be in the production name.
  \ifthenelse{\equal{\pgfkeysvalueof{/bnf/oneof}}{true}}%
    {\newcommand{\oneof}{~~~one~of}}
    {\newcommand{\oneof}{}}
  % Retrieve the label name
  \edef\mylabel{\pgfkeysvalueof{/bnf/term}}%
  % Start of the environment content.
  \begin{table}[H]
  \captionsetup{justification=raggedright,labelformat=empty,skip=0pt}
  \caption{\hspace{\leftmargini}\parbox{\textwidth}{\terminal{\pgfkeysvalueof{/bnf/term}::}\oneof}}%
  % Check if "one of" should be in the production name.
  \ifthenelse{\equal{\pgfkeysvalueof{/bnf/noref}}{true}}%
    {}
    {\label{grammar:\mylabel}}
  \hspace{2\leftmargini}
  \begin{tabular}{@{}#2}
}{%
  \end{tabular}
  \end{table}
  \vspace{-\baselineskip}
}

% The BNF environment will be placed in the appendex of the spec.
\NewDocumentEnvironment{bnf}{ O{term=missing,oneof=false} m }
{%
  \begin{bnfbase}[#1,noref=false]{#2}
}{%
  \end{bnfbase}
}

% This environment will NOT be placed in the appendex of the spec.
\NewDocumentEnvironment{bnfnoindex}{ O{term=missing,oneof=false} m }
{%
  \begin{bnfbase}[#1,noref=true]{#2}
}{%
  \end{bnfbase}
}
